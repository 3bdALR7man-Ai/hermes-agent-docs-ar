      تطوير TUI و Desktop من Worktrees

      تعمل نواة Python بلا مشكلات من أي git worktree — نفّذ cd للدخول إليه وسيعمل hermes مباشرةً. أما الواجهتان المكتوبتان بـ TypeScript فلا تعملان بهذه البساطة: فكلٌّ من ui-tui/ وapps/desktop/ تحتاج إلى مجلد node_modules مُعبّأ، وتنفيذ npm ci من جديد لكل worktree بطيء ويكرّر عدة جيجابايتات عبر كل branch سحبته.

      htui وhgui دالّتان مساعدتان للـ shell تسدّان هذه الفجوة. تُطلق كلٌّ منهما واجهتها من الـ worktree الحالي مع استعارة node_modules من نسخة مسحوبة (checkout) مرجعية واحدة — بحيث لا يكلّفك branch مؤقت سوى رابط رمزي (symlink)، لا عملية تثبيت كاملة.

      هاتان أداتان لتسهيل عمل المطوّرين، وليستا أوامر ضمن الإصدار الرسمي. ضعهما في ~/.zshrc، وعدّل المسارات بما يناسبك.

      نموذج مشاركة الاعتماديات​

      إحدى النسخ المسحوبة هي نسخة الاعتماديات (deps checkout) — المكان الوحيد الذي تشغّل فيه npm install فعليًا. ترتبط كل worktrees الأخرى بها، ولا تعيد التثبيت محليًا إلا حين يختلف ملف القفل (lockfile) الخاص بها (فالـ branch الذي يرفع إصدار إحدى الاعتماديات يجب ألا يعمل بصمت على حزم قديمة).

      يحدّد متغيّرا بيئة النسخة المسحوبة المرجعية:

      المتغيرالمعنى
      HERMES_MAIN_CHECKOUTنسخة الاعتماديات — حيث يوجد node_modules فعليًا، والتي يشغّل الملف .venv/bin/python الخاص بها الـ backend.
      HERMES_GUI_DEPS_CHECKOUTمكان وجود اعتماديات تطبيق سطح المكتب (apps/desktop/node_modules). قيمته الافتراضية هي HERMES_MAIN_CHECKOUT؛ ولا تغيّره إلا إذا كنت تحتفظ باعتماديات سطح المكتب في مكان آخر.

      لا يقرأ Hermes نفسه أيًّا منهما — فهما خاصّان بهاتين الدالتين المساعدتين. أما المتغيرات التي يقرؤها Hermes فعلًا فتجدها في متغيرات البيئة (Environment Variables).

      htui — تشغيل TUI من الـ worktree​

      تملك واجهة Ink TUI مسارًا للتطوير بالفعل: إذ يشغّل hermes --tui --dev مصادر TypeScript عبر tsx بدلًا من الحزمة المبنية مسبقًا. وhtui مجرد سطر واحد فوقه يوجّه التشغيل أيضًا إلى مجلد ui-tui/ في الـ worktree الحالي:

      htui() {
        local root
        root="$(_hermes_root)" || { echo "htui: not in a Hermes checkout" >&2; return 1; }
        ( cd "$root" && PYTHONPATH="$root" \
            "$HERMES_MAIN_CHECKOUT/.venv/bin/python" -m hermes_cli.main --tui --dev "$@" )
      }

      يُترجم --dev من المصدر، ولذلك يربط ui-tui/node_modules من HERMES_MAIN_CHECKOUT حين يتطابق ملف القفل الجذري، ويثبّت محليًا في غير ذلك (راجع الدالة _hermes_root / دوال الربط المساعدة).

      --dev و HERMES_TUI_DIR لا يجتمعان

      يوجّه HERMES_TUI_DIR تطبيق Hermes إلى حزمة مبنية مسبقًا (Nix، حزم النظام)، لا تحتوي على مصدر يمكن إعادة تحميله فوريًا (hot-reload). إذا كان مضبوطًا في الـ shell لديك، فسيخرج hermes --tui --dev برسالة خطأ. نفّذ unset HERMES_TUI_DIR قبل htui.

      hgui — تشغيل تطبيق سطح المكتب من الـ worktree​

      تطبيق سطح المكتب أثقل: فهو يحتاج إلى node_modules في جذر المستودع وفي apps/desktop/ معًا، وإلى خادم تطوير Vite مثبّت على المنفذ 5174، وإلى backend مكتوب بـ Python. تربط hgui كل ذلك بالـ worktree الحالي:

      hgui() {
        local root deps desktop
        root="$(_hermes_root)" || { echo "hgui: not in a Hermes checkout" >&2; return 1; }
        deps="${HERMES_GUI_DEPS_CHECKOUT:-$HERMES_MAIN_CHECKOUT}"
        desktop="$root/apps/desktop"

        # Borrow deps when locks match; otherwise install locally in the worktree.
        if cmp -s "$root/package-lock.json" "$deps/package-lock.json"; then
          _hermes_link_deps "$desktop" "$deps/apps/desktop"
          _hermes_link_deps "$root" "$deps"
        else
          ( cd "$root" && npm ci ) || return 1
        fi

        # Vite is fixed at 5174 — evict a stale session from another hgui.
        lsof -t -i:5174 >/dev/null 2>&1 && killport 5174

        # Electron often survives Ctrl+C without reaping its ephemeral backends.
        trap '_hermes_gui_cleanup "$root"' INT TERM EXIT

        ( cd "$desktop"
          export PATH="$root/node_modules/.bin:$PATH"
          HERMES_DESKTOP_HERMES_ROOT="$root" \
          HERMES_DESKTOP_PYTHON="$HERMES_MAIN_CHECKOUT/.venv/bin/python" \
          HERMES_DESKTOP_IGNORE_EXISTING=1 \
          HERMES_DESKTOP_CWD="$root" \
          npm run dev )
      }

      متغيرات البيئة الخاصة بسطح المكتب التي تضبطها كلها مفاتيح حقيقية لتحديد الـ backend:

      المتغيردوره في hgui
      HERMES_DESKTOP_HERMES_ROOTيشغّل الـ backend من هذا الـ worktree، لا من hermes المُحزَّم أو الموجود في PATH.
      HERMES_DESKTOP_PYTHONيعيد استخدام الـ venv الخاص بنسخة الاعتماديات بدلًا من تحديد Python من جديد.
      HERMES_DESKTOP_IGNORE_EXISTINGيتجاهل أي hermes موجود في PATH كي لا يحجب الـ worktree.
      HERMES_DESKTOP_CWDيفتح محادثة سطح المكتب بحيث يكون جذرها الـ worktree.

      مأزقان تتعامل معهما hgui ولا يتعامل معهما npm run dev المجرد:

      • المنفذ 5174 ثابت. يتصادم تشغيل ثانٍ لـ hgui مع خادم Vite الخاص بالتشغيل الأول؛ لذا تُنهي الدالة المساعدة الخادم القديم أولًا.
      • العمليات الابنة اليتيمة. كثيرًا ما تنجو Electron من Ctrl+C عبر concurrently دون إنهاء الـ backend المؤقت dashboard --port 0 أو عملية Vite. يشغّل الـ trap الخاص بـ EXIT/INT/TERM عملية تنظيف تُنهي غلاف Electron، والمستمع على :5174، وأي لوحة تحكم --port 0 أطلقتها.

      الدوال المساعدة المشتركة​

      تحدّد الدالتان النسخة المسحوبة المحيطة وتربطان الاعتماديات بالطريقة نفسها:

      # The enclosing worktree, verified as a real Hermes checkout.
      _hermes_root() {
        local root
        root="$(git rev-parse --show-toplevel 2>/dev/null)" || return 1
        [[ -f "$root/hermes_cli/main.py" && -d "$root/ui-tui" ]] && print -r "$root"
      }

      # Symlink node_modules from the deps checkout — never over an existing tree.
      _hermes_link_deps() {
        local target="${1%/}" source="${2%/}"
        [[ -d "$source/node_modules" ]] || return 1
        [[ -e "$target/node_modules" ]] || ln -s "$source/node_modules" "$target/node_modules"
      }

      # Reap ephemeral backends Electron leaves behind on exit.
      _hermes_gui_cleanup() {
        local root="$1"
        [[ -n "$root" ]] && pkill -TERM -f "${root}/apps/desktop/node_modules/electron" 2>/dev/null
        lsof -t -i:5174 >/dev/null 2>&1 && killport 5174
        pgrep -f 'hermes_cli\.main.*dashboard.*--port 0' 2>/dev/null | xargs -r kill -TERM 2>/dev/null
      }

      killport دالة مساعدة صغيرة خاصة بك (lsof -ti:$1 | xargs kill)؛ استبدلها بالصيغة التي تفضّلها.

      لماذا يُربط فقط حين تتطابق ملفات القفل

      الرابط الرمزي إلى node_modules مختلف أسوأ من عدم التثبيت أصلًا — إذ سيُبنى الـ worktree على حزم لم يُصرّح بها ملف القفل الخاص به قط. المقارنة بايتًا ببايت لملف package-lock.json هي الحماية الرخيصة والدقيقة: ملف قفل مطابق ⇒ الاستعارة آمنة؛ ملف قفل مختلف ⇒ npm ci محليًا. يحوّل Vite الروابط الرمزية إلى مساراتها الحقيقية (realpath) قبل تطبيق server.fs.allow، ولهذا يُدرج apps/desktop/vite.config.ts الموقع الحقيقي لـ node_modules في القائمة المسموح بها.
